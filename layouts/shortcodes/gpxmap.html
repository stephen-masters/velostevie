{{- $isSection := eq .Page.Kind "section" -}} {{- $gpxFiles := .Page.Resources.Match "*.gpx" -}} {{- /* On section pages, aggregate GPX files from all child articles */ -}} {{- if $isSection -}} {{- range .Page.RegularPages.ByDate -}} {{- range .Resources.Match "*.gpx" -}} {{- $gpxFiles = $gpxFiles | append . -}} {{- end -}} {{- end -}} {{- end -}} {{- /* On single pages, build pre-processed photo marker list from gallery param */ -}} {{- $dir := .Get "gallery" -}} {{- $photoMarkers := slice -}} {{- if and (not $isSection) $dir -}} {{- $gpsData := index $.Site.Data "photo-gps" -}} {{- $images := resources.Match (printf "%s/*" $dir) -}} {{- range $images -}} {{- $filename := path.Base .Name -}} {{- if not (hasPrefix $filename ".") -}} {{- $key := strings.TrimLeft "/" .Name -}} {{- $gps := index $gpsData $key -}} {{- if $gps -}} {{- $full := .Resize "1920x webp" -}} {{- $base := strings.TrimSuffix (path.Ext $filename) $filename -}} {{- $noPrefix := replaceRE "^[0-9]+" "" $base -}} {{- $trimmed := strings.Trim $noPrefix "_" -}} {{- $caption := replace $trimmed "_" " " -}} {{- $marker := dict "url" $full.Permalink "lat" $gps.lat "lng" $gps.lng "caption" $caption -}} {{- $photoMarkers = $photoMarkers | append $marker -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if or $gpxFiles $photoMarkers -}} {{- $urls := slice -}} {{- range $gpxFiles -}} {{- $urls = $urls | append .Permalink -}} {{- end -}}
